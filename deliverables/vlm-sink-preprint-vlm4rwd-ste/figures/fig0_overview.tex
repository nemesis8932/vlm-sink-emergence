% Figure 1: setup. Compiled inside the paper, so it uses the paper's own font (Times) and
% needs no font embedding. Arm colours come from the preamble (\armbaseline etc.), identical
% to analysis/fig_common.py. Two rows: the pipeline on top (inputs -> tokens -> decoder),
% the probe and the lever list underneath.
\begin{tikzpicture}[
  >=stealth, font=\footnotesize, line width=0.5pt,
  box/.style={draw=black!75, rounded corners=2pt, align=center, inner sep=4pt,
              minimum height=2.2em},
  input/.style={box, fill=black!4, draw=black!30, minimum width=3.2em},
  tok/.style={draw=white, line width=0.4pt, minimum size=5.5pt, inner sep=0pt},
  lab/.style={font=\scriptsize, text=black!60},
]
% ------------------------------------------------------------------ row 1: pipeline
\node[input]                                      (img) at (0, 0.90) {image};
\node[input]                                      (txt) at (0, 0)    {text};
\node[box, right=4mm of img, minimum width=6.4em] (enc)
      {SigLIP-B/16\\[-2pt]{\scriptsize pretrained, trainable}};
\node[box, right=4mm of txt, minimum width=6.4em] (tkz) {tokenizer};
\draw[->] (img) -- (enc);
\draw[->] (txt) -- (tkz);

\coordinate (t0) at (4.55, 0.31);
\foreach \i in {0,...,10} {
  \ifnum\i=0
    \node[tok, fill=armtextinit]     (tok\i)  at ($(t0) + (\i*0.205, 0)$) {};
  \else
    \node[tok, fill=armbaseline!45]  (tok\i)  at ($(t0) + (\i*0.205, 0)$) {};
  \fi
}
\foreach \i in {0,...,5} {
    \node[tok, fill=black!22]        (ttok\i) at ($(t0) + (2.42 + \i*0.205, 0)$) {};
}
\node[lab, above=1pt of tok5]  {49 image tokens};
\node[lab, above=1pt of ttok2] {79 text positions};
\draw[->, black!60] (enc.east) -- ++(2.5mm,0) |- ($(tok0.west) + (-1mm, 1.2pt)$);
\draw[->, black!60] (tkz.east) -- ++(2.5mm,0) |- ($(tok0.west) + (-1mm,-1.2pt)$);

\node[box, right=5mm of ttok5, minimum width=8.4em, minimum height=4.6em, anchor=west] (dec)
      {\textbf{decoder}\\[-1pt]{\scriptsize SmolLM2-135M architecture}\\[-2pt]
       {\scriptsize 30 layers, GQA 9q/3kv}};
\draw[->] (ttok5.east) -- (dec.west |- ttok5.east);

% position-0 callout, under the strip
\node[font=\scriptsize\bfseries, text=armtextinit, anchor=north west, inner sep=1pt]
      (p0) at ($(tok0.south west) + (-0.03, -0.20)$) {position 0 = first image token, no BOS};
\draw[armtextinit, line width=0.4pt] (tok0.south) -- ($(tok0.south) + (0, -0.18)$);

% ------------------------------------------------------------------ row 2: probe + levers
\node[box, draw=armtextinit, dashed, fill=armtextinit!6, anchor=north west, align=left,
      inner sep=5pt, font=\scriptsize]
      (probe) at ($(txt.south west) + (0, -0.55)$) {%
  \textbf{probe at position 0, every 100 optimizer steps}\\[2pt]
  \begin{tabular}{@{}l@{\hspace{5pt}}l@{\hspace{6pt}}r@{}}
    concentration & $\mathrm{Sink}^{\epsilon}_1$: share of heads with attn$\to$pos0 $>\epsilon$ & 270 heads\\
    v-ratio & layer-averaged value-norm ratio (Section 3.3) & 30 layers\\
    h-ratio & layer-averaged residual-norm ratio (Section 3.3) & 30 layers
  \end{tabular}};
\draw[->, armtextinit, line width=0.4pt]
      (p0.west) ++(-0.06, 0) -- ++(-0.30, 0) |- ($(probe.north east) + (-0.35, 0)$);

\node[anchor=north west, align=left, font=\scriptsize, inner sep=0pt]
      (arms) at ($(probe.north east) + (0.45, -0.02)$) {%
  \textbf{attention and initialization conditions}\\[1pt]
  \tikz[baseline=-0.5ex]\fill[armbaseline] (0,0) rectangle (5pt,5pt);\ \textit{baseline}: softmax\\
  \tikz[baseline=-0.5ex]\fill[armg1gate]   (0,0) rectangle (5pt,5pt);\ \textit{g1gate}: softmax + zero-init $\sigma$-gate on the output\\
  \tikz[baseline=-0.5ex]\fill[armsigmoid]  (0,0) rectangle (5pt,5pt);\ \textit{sigmoid}: unnormalized sigmoid, no softmax\\
  \tikz[baseline=-0.5ex]\fill[armtextinit] (0,0) rectangle (5pt,5pt);\ \textit{textinit}: decoder loaded from SmolLM2\\
  \tikz[baseline=-0.5ex]\fill[armrf]       (0,0) rectangle (5pt,5pt);\ \textit{RF}: baseline recipe, fresh 1B-token stream%
};
\end{tikzpicture}
